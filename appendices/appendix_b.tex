\section{Class Imbalance and Resampling Details}
% ============================================================

\paragraph{Class weighting.}
Balanced class weighting modifies the learning objective so that mistakes
on minority classes receive larger penalties. The balanced weight assigned
to class \(c\) is

\begin{equation}
\label{eq:class_weight}
w_c =
\frac{N}{K n_c},
\end{equation}

where \(N\) is the total number of training samples, \(K\) is the number
of classes, and \(n_c\) is the number of training samples in class \(c\).
Using the training counts in Chapter~\ref{ch:method}, the approximate
balanced class weights are:

\[
w_{\text{MAIZE}} \approx 0.44,\quad
w_{\text{RICE}} \approx 0.71,\quad
w_{\text{BEAN}} \approx 5.37,\quad
w_{\text{IRISH POTATO}} \approx 6.82.
\]

For models without a native \texttt{class\_weight} interface in this
workflow, the same principle is implemented through per-sample weights:
\begin{equation}
\label{eq:sample_weight}
s_i = w_{y_i},
\end{equation}
where \(s_i\) is the sample weight of observation \(i\), and \(w_{y_i}\)
is the balanced class weight of its true class.

\paragraph{SMOTE.}
For a minority-class sample \(x_i\) and one of its \(k\)-nearest minority
neighbours \(x_j\), SMOTE generates a synthetic sample by interpolation:

\begin{equation}
\label{eq:smote_formula}
x_{\mathrm{new}}
=
x_i + \lambda (x_j - x_i),
\quad
\lambda \sim U(0,1).
\end{equation}

In this project, SMOTE is applied with \(k=5\) nearest neighbours. The
targeted per-class SMOTE strategies use RICE as the reference class rather
than MAIZE. With \(6{,}194\) RICE training samples, the four tested targets
are:

\[
\begin{aligned}
\text{light target}  &= 0.4 \times 6{,}194 \approx 2{,}478,\\
\text{medium target} &= 0.6 \times 6{,}194 \approx 3{,}716,\\
\text{strong target} &= 0.8 \times 6{,}194 \approx 4{,}955,\\
\text{rice-level target} &= 1.0 \times 6{,}194 = 6{,}194.
\end{aligned}
\]

Under the selected per-class SMOTE medium setting, BEAN and IRISH POTATO
are each oversampled to approximately \(3{,}716\) samples:

\[
\begin{aligned}
\text{BEAN synthetic samples} &\approx 3{,}716 - 814 = 2{,}902,\\
\text{IRISH POTATO synthetic samples} &\approx 3{,}716 - 641 = 3{,}075.
\end{aligned}
\]

\paragraph{ADASYN.}
ADASYN estimates local classification difficulty around each minority
sample \(x_i\) from its nearest-neighbour region:

\begin{equation}
\label{eq:adasyn_difficulty}
r_i =
\frac{\Delta_i}{k},
\end{equation}

where \(\Delta_i\) is the number of majority-class samples among the
\(k\)-nearest neighbours of \(x_i\). The difficulty values are normalised as

\begin{equation}
\label{eq:adasyn_normalised}
\hat{r}_i =
\frac{r_i}{\sum_i r_i}.
\end{equation}

The number of synthetic samples generated around minority sample \(x_i\) is

\begin{equation}
\label{eq:adasyn_generation}
g_i =
\hat{r}_i G,
\end{equation}

where \(G\) is the total number of synthetic samples to be generated.

\paragraph{Weighted sequence loss.}
For sequence models, class imbalance is handled through class weights on
the original temporal sequences. The weighted sparse categorical
cross-entropy is:

\begin{equation}
\label{eq:weighted_sequence_loss}
\mathcal{L}_{\mathrm{weighted}}
=
-\frac{1}{N}
\sum_{i=1}^{N}
w_{y_i}
\log p_{i,y_i},
\end{equation}

where \(y_i\) is the true class of sample \(i\), \(w_{y_i}\) is the weight
of that class, and \(p_{i,y_i}\) is the predicted probability assigned to
the true class.

\section{Evaluation Metric Definitions}
% ============================================================

\paragraph{Macro F1.}
For each class \(c\), the F1 score is:
\begin{equation}
F1_c = \frac{2 \cdot P_c \cdot R_c}{P_c + R_c}
\end{equation}
where \(P_c\) is precision and \(R_c\) is recall. Macro F1 is the
unweighted mean across all classes:
\begin{equation}
F1_{\mathrm{macro}} = \frac{1}{K} \sum_{c=1}^{K} F1_c .
\end{equation}

\paragraph{Weighted F1.}
\begin{equation}
F1_{\mathrm{weighted}} =
\frac{\sum_{c=1}^{K} n_c \cdot F1_c}{\sum_{c=1}^{K} n_c},
\end{equation}
where \(n_c\) is the number of true samples of class \(c\).

\paragraph{Balanced Accuracy.}
\begin{equation}
\mathrm{BA} = \frac{1}{K} \sum_{c=1}^{K} R_c .
\end{equation}

\paragraph{Cohen's Kappa.}
\begin{equation}
\kappa = \frac{p_o - p_e}{1 - p_e},
\end{equation}
where \(p_o\) is observed agreement and \(p_e\) is expected agreement by
chance.

\paragraph{AUC metrics.}
ROC-AUC is computed using a one-vs-rest macro-averaged receiver operating
characteristic curve. PR-AUC is computed per class from the
precision--recall curve and averaged across classes; it is especially
informative for minority-class performance because it does not reward the
large true-negative pool in imbalanced datasets.
