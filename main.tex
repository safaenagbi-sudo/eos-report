% ── Chapter ──
\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{chaptercolor}}
  {%
    \begin{tikzpicture}
      \draw[color=accentgold, line width=1.2pt] (0,0) -- (2.5,0);
      \draw[color=accentgold, line width=1.2pt] (0,0) -- (0,-0.5);
      \node[right] at (2.6,0)
        {\normalfont\large\color{chaptercolor}\textsc{Chapter}~\thechapter};
    \end{tikzpicture}%
  }
  {0pt}
  {%
    \vspace{4pt}%
    \begin{tcolorbox}[
      enhanced, colback=accentgold!10, colframe=accentgold,
      arc=4pt, boxrule=0.8pt, left=10pt, right=10pt, top=6pt, bottom=6pt,
      shadow={2pt}{-2pt}{0pt}{gray!30}
    ]
      \centering\LARGE\bfseries\color{chaptercolor}#1
    \end{tcolorbox}%
  }
\titlespacing*{\chapter}{0pt}{-20pt}{20pt}
